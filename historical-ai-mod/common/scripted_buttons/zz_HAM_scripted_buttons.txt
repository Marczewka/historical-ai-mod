						}
					}
				}

			}
		}
		custom_tooltip = {
			text = risorgimento_unifying_stance_tt
			has_variable = risorgimento_unifying_stance
		}
	}

	ai_chance = {
		value = 50
		if = {
			limit = {
				is_at_war = yes
			}
			add = -100
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_theocracy
			}
			add = -70
		}
		if = {
			limit = {
				any_country = {
					NOR = {
						country_has_primary_culture = cu:south_italian
						country_has_primary_culture = cu:north_italian
					}
					any_scope_state = {
						state_region = {
							OR = {
								is_homeland = cu:south_italian
								is_homeland = cu:north_italian
							}
						}
					}
					relations:root <= 20
				}
			}
			add = 25
		}
	}

	effect = {
		ROOT = {
			save_scope_as = saboteur_country
		}
		every_country = {
			limit = {
				NOR = {
					country_has_primary_culture = cu:south_italian
					country_has_primary_culture = cu:north_italian
				}
				any_scope_state = {
					state_region = {
						OR = {
							is_homeland = cu:south_italian
							is_homeland = cu:north_italian
						}
					}
				}
			}
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:north_italian
					}
				}
				if = {
					limit = {
						ROOT = {
							country_rank >= rank_value:major_power
						}
					}
					add_radicals_in_state = {
						value = 0.10
						culture = cu:north_italian
					}
				}
				else = {
					add_radicals_in_state = {
						value = 0.05
						culture = cu:north_italian
					}
				}
			}
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:south_italian
					}
				}
				if = {
					limit = {
						ROOT = {
							country_rank >= rank_value:major_power
						}
					}
					add_radicals_in_state = {
						value = 0.10
						culture = cu:south_italian
					}
				}
				else = {
					add_radicals_in_state = {
						value = 0.05
						culture = cu:south_italian
					}
				}
			}
			post_notification = nationalism_decision_agitation_occurred
		}
		je:je_risorgimento ?= {
			add_modifier = {
				name = support_separatism_nationalist_propaganda
				multiplier = 5
				days = 1095
			}
		}
		custom_tooltip = ausrad_cooldown_year_tt
	}

				NOR = { 
					has_law_or_variant = law_type:law_autocracy 
					has_law_or_variant = law_type:law_oligarchy
				}
			}
			add = 50
		}

				OR = { 
					has_law_or_variant = law_type:law_census_voting
					has_law_or_variant = law_type:law_universal_suffrage
				}
			}
			add = 50
		}
		if = {
			limit = {
				NOT = { 
					has_law_or_variant = law_type:law_traditionalism
				}
			}
			add = 50
		}
	}

	effect = {
		root = {
			save_scope_as = italian_unifier_candidate
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_patriotic
					has_ideology = ideology:ideology_liberal

						OR = {						
							has_ideology = ideology:ideology_radical
							has_ideology = ideology:ideology_jingoist_leader
						}
					}
				}
			}
			root = {
				if = {
					limit = {
						country_has_primary_culture = cu:south_italian
					}

					}
				}

					}
					add_loyalists = {
						value = medium_radicals
						culture = cu:north_italian
						interest_group = prev
					}
				}
			}
			add_modifier = {
				name = modifier_risorgimento_supporters
				months = normal_modifier_time
			}
		}
		every_political_movement = {
			limit = {
				has_modifier = modifier_italian_unificationist_sentiment
			}
			remove_modifier = modifier_italian_unificationist_sentiment
		}
		set_variable = risorgimento_unifying_stance
		if = {
			limit = {
				has_variable = risorgimento_anti_unifying_stance
			}
			remove_variable = risorgimento_anti_unifying_stance
		}
		add_modifier = {
			name = modifier_risorgimento_support

		}
		if = {
			limit = {
				has_modifier = modifier_risorgimento_opposition
			}
			remove_modifier = modifier_risorgimento_opposition
		}
		every_country = {
			limit = {
				has_interest_marker_in_region = sr:region_southern_europe
				NOT = {
					THIS = ROOT
				}
			}

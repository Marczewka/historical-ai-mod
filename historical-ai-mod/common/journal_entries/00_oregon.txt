je_oregon = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_usa_manifest_destiny

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		exists = c:USA
		exists = c:GBR
		c:USA = ROOT
		s:STATE_WASHINGTON = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
		s:STATE_OREGON = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
		s:STATE_IDAHO = {
			any_scope_state = {
				owner = {
					OR = {
						c:GBR = THIS
						is_subject_of = c:GBR
					}
				}
			}
		}
	}

	possible = {
		always = yes
		# OR = {
		# 	has_interest_marker_in_region = sr:region_pacific_coast
		# 	custom_tooltip = {
		# 		text = oregon_tt
		# 		has_global_variable = oregon_trail_mapped
		# 	}
		# 	owns_entire_state_region = STATE_MONTANA
		# 	owns_entire_state_region = STATE_WYOMING
		# }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			c:GBR ?= {
				relations:ROOT > relations_threshold:poor
			}
			has_technology_researched = nationalism
			any_scope_state = {
				OR = {
					state_region = s:STATE_MONTANA
					state_region = s:STATE_WYOMING
				}
			}
		}

		effect = {
			c:GBR ?= { trigger_event = { id = oregon.2 } }
			je_oregon_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			c:GBR = {
				relations:ROOT <= relations_threshold:cold
			}
		}
		effect = {
			c:USA = {
				trigger_event = {
					id = oregon.4
					days = 0
				}
			}
			je_oregon_cleanup = yes
		}
	}

	# Changed direct acquisition from complete to invalid state
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			any_subject_or_below = {
				AND = {
					owns_entire_state_region = STATE_WASHINGTON
					owns_entire_state_region = STATE_IDAHO
					owns_entire_state_region = STATE_OREGON
				}
			}
		}
		effect = {
			s:STATE_IDAHO = {
				add_homeland = cu:yankee
			}
			s:STATE_WASHINGTON = {
				add_homeland = cu:yankee
			}
			s:STATE_OREGON = {
				add_homeland = cu:yankee
			}
			if = {
				limit = {
					s:STATE_BRITISH_COLUMBIA = {
						any_scope_state = {
							owner = {
								OR = {
									c:USA ?= THIS
									is_subject_of = c:USA
								}
							}
						}
					}
				}
				s:STATE_BRITISH_COLUMBIA = {
					add_homeland = cu:yankee
				}
			}
			add_modifier = {
				name = oregon_resolved
				days = normal_modifier_time
			}
			je_oregon_cleanup = yes
		}
	}

	weight = 100
}
